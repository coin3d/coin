cmake_minimum_required(VERSION 3.15)
project(CoinPickedPointNormalTransform LANGUAGES CXX)
find_package(Coin CONFIG REQUIRED)
add_executable(picked-point-normal-transform main.cpp)
target_compile_features(picked-point-normal-transform PRIVATE cxx_std_11)
target_link_libraries(picked-point-normal-transform PRIVATE Coin::Coin)
enable_testing()
foreach(group world hierarchy controls ownership)
  add_test(NAME picked-point-normal-${group}
           COMMAND picked-point-normal-transform ${group})
endforeach()
