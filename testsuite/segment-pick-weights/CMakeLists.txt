cmake_minimum_required(VERSION 3.15)
project(CoinSegmentPickWeights LANGUAGES CXX)
find_package(Coin CONFIG REQUIRED)
include(CTest)
add_executable(segment-pick-weights main.cpp)
target_compile_features(segment-pick-weights PRIVATE cxx_std_11)
target_link_libraries(segment-pick-weights PRIVATE Coin::Coin)

  add_test(NAME segment-pick-${group} COMMAND segment-pick-weights ${group})
endforeach()
