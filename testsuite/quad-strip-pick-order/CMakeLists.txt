cmake_minimum_required(VERSION 3.15)
project(CoinQuadStripPickOrder LANGUAGES CXX)
find_package(Coin CONFIG REQUIRED)
include(CTest)
add_executable(quad-strip-pick-order main.cpp)
target_compile_features(quad-strip-pick-order PRIVATE cxx_std_11)
target_link_libraries(quad-strip-pick-order PRIVATE Coin::Coin)
foreach(scenario generic-strip generic-quads generic-triangles detail-copy mesh-state mesh-vertex-property)
  add_test(NAME ${scenario} COMMAND quad-strip-pick-order ${scenario})
endforeach()
