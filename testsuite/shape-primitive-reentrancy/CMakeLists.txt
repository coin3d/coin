cmake_minimum_required(VERSION 3.16)
project(shape_primitive_reentrancy LANGUAGES CXX)
find_package(Coin CONFIG REQUIRED)
enable_testing()
add_executable(shape_primitive_reentrancy reentrancy.cpp)
target_link_libraries(shape_primitive_reentrancy PRIVATE Coin::Coin)
target_compile_features(shape_primitive_reentrancy PRIVATE cxx_std_11)
foreach(scenario controls different same deep count pick count_callback pick_callback partial polygon convex abort prune exception empty single)
  add_test(NAME shape_reentrancy_${scenario} COMMAND shape_primitive_reentrancy ${scenario})
  set_tests_properties(shape_reentrancy_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
